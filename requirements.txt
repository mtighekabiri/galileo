# LUMEN Insertion Tool runtime dependencies.

# opencv-python-headless rather than opencv-python: the full build ships its
# own Qt platform plugin (libqxcb.so) which clashes with PyQt5's on Linux and
# produces the classic "could not load the Qt platform plugin xcb" failure.
# This app never calls cv2's GUI functions, so the headless build loses nothing.
opencv-python-headless>=4.8
PyQt5>=5.15,<6
numpy>=1.24
